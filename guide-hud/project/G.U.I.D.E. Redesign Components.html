01Carry ModuleEquipment › right column, below 04 Gear

A visual sibling of the shard panel, and deliberately so: shards extend what the character can do, containers extend what they can hold. Both are system extensions, so both are framed the same way. Equip and unequip live here and nowhere else — a container's tab on the Inventory screen is a consequence of this list, never its own control.

 
Page rowSummary + tab-arrow — contents live in an Inventory tab (field pack, bag of holding, sack)
Inline rowCyan capacity + chevron — contents expand in place, never get a tab (quiver, bolt case, scroll case)
Abstract rowStatus word only, nothing to open (belt pouch)
StowedOwned-but-unequipped containers collapse onto one line, so stowing never grows the panel
EmptyHatched strip that reads as an equip target rather than a blank panel

  function ammoStacks() {
    const q = CARRY.find(c => c.id === 'quiver');
    return q && q.equipped ? q.contents : [];
  }
  const ammoCount = () => ammoStacks().reduce((s, a) => s + a.q, 0);
  let activeAmmo = 0;

  function carryRowHtml(c) {
    const read =
      c.type === 'inline'   ? '<span class="s-read"><span class="cap">' + ammoCount() + ' / ' + c.cap + '</span> Arrows</span>'
    : c.type === 'abstract' ? '<span class="s-read"><span class="status">' + c.status + '</span></span>'
    :                         '<span class="s-read">' + c.summary + ' <i class="fa-solid fa-arrow-up-right-from-square tab-go" title="Opens as a tab in Inventory"></i></span>';
    const chev = c.type === 'inline'
      ? '<button class="s-chev" data-expand="' + c.id + '" aria-label="Show contents"><i class="fa-solid fa-chevron-right"></i></button>' : '';
    const body = c.type === 'inline'
      ? '<div class="crow-body">' + c.contents.map(a =>
          '<div class="ln' + (a.rare ? ' rare-ln' : '') + '"><span>' + a.n + '</span><span class="q">×' + a.q + '</span></div>').join('') + '</div>'
      : '';
    return '<div class="carry-row" data-row="' + c.id + '" data-rarity="' + c.rarity + '">' +
      '<div class="slot" data-rarity="' + c.rarity + '" data-tt=\'' + JSON.stringify(c.tt).replace(/'/g, '&#39;') + '\'>' +
        '<span class="s-frame"></span>' +
        '<span class="s-inner">' +
          '<span class="s-icon"><i class="fa-solid ' + c.icon + '"></i></span>' +
          '<span class="s-nm">' + c.name + '</span>' +
          read + chev +
          '<button class="c-act" data-toggle="' + c.id + '">Unequip</button>' +
        '</span>' +
        '<span class="rarity-dot"></span>' +
      '</div>' + body + '</div>';
  }

  /* owned but unequipped: one line for all of them, so the panel never
     grows just because the character kept a sack they are not carrying */
  function stowedHtml(list) {
    return '<div class="carry-stowed"><span class="k">Stowed</span>' +
      list.map(c => '<span class="nm">' + c.name + '</span>').join('<span class="k">·</span>') +
      '<span class="sp"></span>' +
      list.map(c => '<button class="c-act equip" data-toggle="' + c.id + '">Equip</button>').join('') +
      '</div>';
  }

  const targetHtml =
    '<div class="carry-target" data-toggle="open" title="Equip a container from the inventory"><i class="fa-solid fa-plus"></i>' +
    '<span>Equip a Container</span><span class="sp"></span><span class="hint">Adds Storage</span></div>';

  function renderCarry() {
    const host = document.getElementById('carry-mod');
    const open = [...host.querySelectorAll('.carry-row.open')].map(r => r.dataset.row);
    const on = CARRY.filter(c => c.equipped), off = CARRY.filter(c => !c.equipped);
    host.innerHTML = on.map(carryRowHtml).join('') +
      (off.length ? stowedHtml(off) : '') + targetHtml +
      '<div class="carry-foot"><span class="acc">◇</span> Containers extend storage <span style="color:var(--beige-dim)">·</span> shards extend capability</div>';
    open.forEach(id => host.querySelector('[data-row="' + id + '"]')?.classList.add('open'));
    host.querySelectorAll('[data-tt]').forEach(bindTooltip);
    host.querySelectorAll('[data-expand]').forEach(b => b.addEventListener('click', e => {
      e.stopPropagation();
      b.closest('.carry-row').classList.toggle('open');
    }));
    host.querySelectorAll('[data-toggle]').forEach(b => b.addEventListener('click', e => {
      e.stopPropagation();
      const id = b.dataset.toggle;
      if (id === 'open') { console.log('Equip a container into the open carry point...'); return; }
      const c = CARRY.find(x => x.id === id);
      c.equipped = !c.equipped;
      console.log((c.equipped ? 'EQUIPPED' : 'UNEQUIPPED') + ': ' + c.name +
        (c.type === 'page' ? ' — ' + (c.equipped ? 'adds' : 'removes') + ' the ' + c.tab.toUpperCase() + ' tab in Inventory' : ''));
      renderCarry(); renderAmmo();
    }));
    document.getElementById('carry-meta').textContent = CARRY.filter(c => c.equipped).length + ' equipped';
  }
02Ammo PickerEquipment › 01 Weapons, ranged cards only

Which arrow is nocked is a property of the attack, not of the container — so the selector sits on the weapon card beside ATTACK, not in the quiver. Melee cards never get one, and if no quiver is equipped the control is absent rather than empty. The menu renders to a fixed layer because the weapon card is clip-pathed and would crop it.

 
SourceStacks come from the equipped quiver row in the carry module above
AttackLogs the weapon plus the nocked type and its remaining count
AbsentUnequip the quiver in block 01 and this control disappears

/* ----- ammo picker: nocked arrow is a property of the attack ----- */
  function renderAmmo() {
    const host = document.getElementById('ammo-recurve');
    if (!host) return;
    const stacks = ammoStacks();
    if (!stacks.length) { host.innerHTML = ''; host.style.display = 'none'; return; }
    host.style.display = '';
    if (activeAmmo >= stacks.length) activeAmmo = 0;
    const a = stacks[activeAmmo];
    host.innerHTML =
      '<button class="ammo-btn"><i class="fa-solid fa-location-arrow"></i><span class="am-name">' + a.n + '</span>' +
      '<span class="am-ct">×' + a.q + '</span><i class="fa-solid fa-chevron-down chev"></i></button>';
    const menu = document.getElementById('ammo-menu');
    menu.innerHTML = stacks.map((s, i) =>
      '<button class="opt ' + (i === activeAmmo ? 'on' : '') + '" data-ammo="' + i + '"><span>' + s.n + '</span><span class="q">×' + s.q + '</span></button>').join('');
    const btn = host.querySelector('.ammo-btn');
    btn.addEventListener('click', e => {
      e.stopPropagation();
      const open = menu.classList.contains('show') && menu.dataset.owner === 'recurve';
      closeAmmo();
      if (open) return;
      menu.dataset.owner = 'recurve';
      menu.classList.add('show');
      host.classList.add('open');
      const r = btn.getBoundingClientRect();
      menu.style.left = r.left + 'px';
      menu.style.top = Math.max(12, r.top - menu.offsetHeight - 6) + 'px';
    });
    menu.querySelectorAll('[data-ammo]').forEach(o => o.addEventListener('click', e => {
      e.stopPropagation();
      activeAmmo = +o.dataset.ammo;
      console.log('Nocked: ' + stacks[activeAmmo].n);
      closeAmmo();
      renderAmmo();
    }));
  }
  function closeAmmo() {
    document.getElementById('ammo-menu').classList.remove('show');
    document.querySelectorAll('.ammo.open').forEach(a => a.classList.remove('open'));
  }
  document.addEventListener('click', closeAmmo);
  document.querySelectorAll('[data-atk]').forEach(b => b.addEventListener('click', e => {
    e.stopPropagation();
    const st = ammoStacks()[activeAmmo];
    const ranged = b.closest('[data-ranged]');
    console.log('ATTACK: ' + b.dataset.atk + (ranged && st ? ' · ' + st.n + ' (' + st.q + ' left)' : ''));
  }));
03Container Switcher & On-Person GridInventory › 01 Carried

The grid is no longer the inventory. ON PERSON is a fixed 5×4 loadout of what the character can physically reach; bulk lives in containers, which are unlimited lists. Carry weight is the only capacity limit and it never blocks a pickup — it only slows. Switching tabs re-renders the pane, the header line and the utility bar only, so nothing around the panel moves.

PointerFine = hover tooltip + centred popup · Coarse = no tooltip, bottom sheet

No quiver tabInline containers are never browsed as a page — arrows are drawn by the weapon's ammo picker
ListsCategory chips + sort; header line carries item count and weight
Locked tabSack is owned but not carried — click is denied, not hidden
UnequipRight of the header line; contents stay with the container
HoverFacts-only tooltip — name, category, rarity, weight, key stat, quantity. No prose, no buttons; it is for scanning
ClickPopup in the shard-modal frame: facts grid, flavour, granted effects, and the actions for that item
ActionsEquip · Stow (with a destination picker when more than one container accepts it) · Retrieve · Drop. Quest items cannot be dropped
RefusalRetrieve with no free cell warns in place instead of failing silently
CoarseTooltip suppressed, popup becomes a swipe-down bottom sheet with 52px targets
